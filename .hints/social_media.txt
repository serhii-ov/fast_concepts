PyJWT – Auth
Docker – Containerization
Pytest – Testing


from fastapi import FastAPI
from app.core.config import settings
from app.api.v1 import routes_user, routes_post

app = FastAPI(title="Social Media Micro API")

app.include_router(routes_user.router, prefix="/api/v1/users", tags=["Users"])
app.include_router(routes_post.router, prefix="/api/v1/posts", tags=["Posts"])


# Install dependencies
pip install -r requirements.txt

# Set up .env file
cp .env.example .env

# Run DB migrations
alembic upgrade head

# Start app
uvicorn app.main:app --reload

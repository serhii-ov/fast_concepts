# app/api/v1/routes_chat.py
from fastapi import APIRouter, WebSocket, Depends
from app.core.websocket import manager
from app.core.security import get_current_user_ws
from app.schemas.user import UserOut

router = APIRouter()

@router.websocket("/ws/chat/{room_id}")
async def chatroom_ws(websocket: WebSocket, room_id: str, user: UserOut = Depends(get_current_user_ws)):
    await manager.connect(websocket, room_id, user.username)
    try:
        while True:
            data = await websocket.receive_text()
            await manager.broadcast(room_id, f"{user.username}: {data}")
    except Exception:
        await manager.disconnect(websocket, room_id)
